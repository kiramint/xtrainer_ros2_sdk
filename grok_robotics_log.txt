import cv2, numpy as np, json, socket

def call(tool, args, timeout=25):
    req = {"tool": tool, "arguments": args, "timeout": timeout}
    s = socket.socket(socket.AF_UNIX, socket.SOCK_STREAM)
    s.settimeout(timeout+10)
    s.connect("/tmp/xtrainer_mcp_bridge.sock")
    s.sendall((json.dumps(req)+"\n").encode())
    buf=b""
    while b"\n" not in buf:
        c=s.recv(1<<20)
        if not c: break
        buf += c
    s.close()
    return json.loads(buf.decode())

def quat_of(alpha_deg):
    a=np.deg2rad(alpha_deg)
    c,s=np.cos(a),np.sin(a)
    # columns: toolX, toolY, toolZ
    R=np.array([
        [c, 0, s],
        [0,-1, 0],
        [s, 0,-c],
    ], dtype=float)
    t=np.trace(R)
    if t>0:
        S=np.sqrt(t+1.0)*2
        w=0.25*S; x=(R[2,1]-R[1,2])/S; y=(R[0,2]-R[2,0])/S; z=(R[1,0]-R[0,1])/S
    elif R[0,0]>R[1,1] and R[0,0]>R[2,2]:
        S=np.sqrt(1+R[0,0]-R[1,1]-R[2,2])*2
        w=(R[2,1]-R[1,2])/S; x=0.25*S; y=(R[0,1]+R[1,0])/S; z=(R[0,2]+R[2,0])/S
    elif R[1,1]>R[2,2]:
        S=np.sqrt(1+R[1,1]-R[0,0]-R[2,2])*2
        w=(R[0,2]-R[2,0])/S; x=(R[0,1]+R[1,0])/S; y=0.25*S; z=(R[1,2]+R[2,1])/S
    else:
        S=np.sqrt(1+R[2,2]-R[0,0]-R[1,1])*2
        w=(R[1,0]-R[0,1])/S; x=(R[0,2]+R[2,0])/S; y=(R[1,2]+R[2,1])/S; z=0.25*S
    q=np.array([x,y,z,w]); q/=np.linalg.norm(q)
    if q[3]<0: q=-q
    return q
